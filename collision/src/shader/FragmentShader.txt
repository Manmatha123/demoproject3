#version 150

in vec2 pass_textureCoordinates;
in vec3 surfaceNormal;
in vec3 toLightVector;
in vec3 tocameravector;
in float visibility;
in vec4 shadowCoords;

out vec4 out_colour;

uniform sampler2D shadowMap;
uniform sampler2D modelTexture;
uniform vec3 lightcolour;
uniform float shinedamper;
uniform float reflectivity;
uniform vec3 skycolour;

void main(void) {

    // Convert from clip space to texture space
    vec3 projCoords = shadowCoords.xyz / shadowCoords.w;
    projCoords = projCoords * 0.5 + 0.5;

    // Basic lighting
    vec3 unitNormal = normalize(surfaceNormal);
    vec3 unitLightVector = normalize(toLightVector);
    float nDotL = max(dot(unitNormal, unitLightVector), 0.0);
    vec3 diffuse = max(nDotL, 0.4) * lightcolour;

    vec3 unitToCamera = normalize(tocameravector);
    vec3 reflectedLight = reflect(-unitLightVector, unitNormal);
    float specFactor = max(dot(reflectedLight, unitToCamera), 0.0);
    float dampedSpec = pow(specFactor, shinedamper);
    vec3 specular = dampedSpec * lightcolour * reflectivity;

    vec4 textureColour = texture(modelTexture, pass_textureCoordinates);

    // If fragment outside shadow map, render normally
    if(projCoords.x < 0.0 || projCoords.x > 1.0 || 
       projCoords.y < 0.0 || projCoords.y > 1.0 || 
       projCoords.z > 1.0) 
    {
        out_colour = vec4(diffuse, 1.0) * textureColour + vec4(specular, 1.0);
        return;
    }

    // Small bias to prevent shadow acne
    float bias = max(0.002 * (1.0 - dot(unitNormal, unitLightVector)), 0.001);

    // Simple 3x3 PCF for soft shadows
    float shadow = 0.0;
    vec2 texelSize = 1.0 / textureSize(shadowMap, 0);
    for(int x = -1; x <= 1; ++x) {
        for(int y = -1; y <= 1; ++y) {
            float closestDepth = texture(shadowMap, projCoords.xy + vec2(x, y) * texelSize).r;
            if(projCoords.z - bias > closestDepth) {
                shadow += 1.0;
            }
        }
    }
    shadow /= 9.0;

    // Apply shadow factor
    diffuse *= 1.0 - 0.9 * shadow;

    // Final color output
    out_colour = vec4(diffuse, 1.0) * textureColour + vec4(specular, 1.0);
}
