//************create dotted line************


public static void drawDottedLine(
        Vector3f start,
        Vector3f end,
        Vector3f color,
        float segmentLength,
        float gapLength) {

    Vector3f dir = new Vector3f(end).sub(start);
    float length = dir.length();
    dir.normalize();

    float step = segmentLength + gapLength;

    for (float d = 0; d < length; d += step) {
        Vector3f a = new Vector3f(dir).mul(d).add(start);
        Vector3f b = new Vector3f(dir)
                .mul(Math.min(d + segmentLength, length))
                .add(start);

        Debug.drawLine(a, b, color);
    }
}
